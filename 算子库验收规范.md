  - 变量名: 小写字母，下划线分隔（snake_case）

  - 常量名: 大写字母，下划线分隔（UPPER_CASE）

1. 精度测试
1. FlagGems 
    - 整体原则：
  计算类型算子的精度将以 CPU-FP64 作为基准，而逻辑类型算子以 CPU int32 为基准。
  对于硬件厂商的某一个算子 F，其功能记为:
  $$x,y,z,... = F_{p}(a,b,c,...)$$
  其中 a，b，c，... 为输入 tensor； p 为被测数制（即数据精度）；x，y，z，... 为各个硬件厂商的输出。
  对于采用 assert close 的算子，我们定义算子在 CPU-FP64 下的计算结果为 Golden Reference，记为 gr，厂商在其硬件下同算子计算得到的结果与 CPU-FP64 下的精度进行对比，对比前我们将 gr 进行数据类型转换成被测数制的数据类型 gr'。
  对于算子的每一个输出，通过精度验证的标准为：
  torch.testing.assert_close(y, gr', atol, rtol) = True
  对于采用assert_eq的算子，定义算子在 CPU-INT32 下的计算结果为 Golden Reference，并调用相应接口判等；对于随机类型的算子，FlagGems仓库针对该算子的分布特性等提供相应单测。
    - 断言与容差设置：
      - 断言：
        - 提供统一的断言函数
          - gems_assert_close(res, ref, dtype, equal_nan=False, reduce_dim=1, atol=1e-4)
          - gems_assert_equal(res, ref, equal_nan=False)
        - 选择的原则：
          - 位精确操作（如 abs、logical_not）：使用 gems_assert_equal
          - 浮点运算（如 exp、sin、matmul）、归约操作（sum、mean 等）：使用 gems_assert_close
      - 容差：
      不同数据类型的数值精度差异较大（如 FP32、BF16、FP16 等），通过集中定义不同 dtype 的容差标准，实现精度评估的一致性、可比性和低维护成本。具体的可以参考 FlagGems 中的实现：
        torch.bool: 0,
        torch.uint8: 0,
        torch.int8: 0,
        torch.int16: 0,
        torch.int32: 0,
        torch.int64: 0,
        torch.float8_e4m3fn: 1e-3,
        torch.float8_e5m2: 1e-3,
        torch.float8_e4m3fnuz: 1e-3,

    - 测试Shape：
      通过统一维护测试 shape，实现测试场景的一致性与可扩展性，提升结果可比性并降低维护成本。具体的可以参考 tests/accuracy_utils.py 。

      - 特殊形状：benchmark/core_shapes.yaml 
      算子加速比的计算规则：加速比采用“两级平均”策略：先按 shape 聚合，再按精度聚合，得到算子整体性能表现。具体来说，每个算子会测试多种精度，每个精度下有多个 shape，对同一精度，多个 shape 求平均作为该精度下的加速比、然后多个精度下的加速比再求平均，作为该算子的加速比。

def ensure_dir(p):
    Path(p).mkdir(parents=True, exist_ok=True)

def now_ts():
    return datetime.datetime.now().strftime("%Y%m%d_%H%M%S")

def run_cmd_capture(cmd, cwd=None, env=None):
    """Run shell command and capture stdout, stderr, exitcode. Returns (out, err, code)."""
    print(f"[INFO] CMD: {cmd}  (cwd={cwd})")
    p = subprocess.Popen(cmd, cwd=cwd, env=env, shell=True,
                         stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True)
    out, err = p.communicate()
    return out or "", err or "", p.returncode

# robust numeric validator: supports decimals and scientific notation, allows leading '.' (e.g. .5)
NUM_RE = re.compile(r'^[+-]?(?:\d+\.?\d*|\.\d+)(?:[eE][+-]?\d+)?$')

def is_number(s):
    return bool(NUM_RE.match(s.strip()))

def to_decimal(s):
    s2 = s.strip()
    if not is_number(s2):
        raise ValueError(f"Not numeric: {s}")
    return Decimal(s2)

# ---------------- Parse pytest summary ----------------

ANSI_RE = re.compile(r'\x1B\[[0-?]*[ -/]*[@-~]')

def parse_pytest_summary_from_text(text):
    """
    Return tuple: (passed, failed, skipped, xfailed, xpassed, errors, total)
    total = passed + failed + skipped
    """
    clean = ANSI_RE.sub('', text)
    counters = {"passed":0, "failed":0, "skipped":0, "xfailed":0, "xpassed":0, "errors":0}

    for m in re.finditer(r'(\d+)\s+([A-Za-z_]+)', clean):
        num = int(m.group(1))
        key = m.group(2).lower()
        if key in counters:
            counters[key] = num

    passed = counters["passed"]
    failed = counters["failed"]
    skipped = counters["skipped"]
    xfailed = counters["xfailed"]
    xpassed = counters["xpassed"]
    errors = counters["errors"]
    total = passed + failed + skipped
    return passed, failed, skipped, xfailed, xpassed, errors, total

def run_accuracy(op, gpu_id, flaggems_path, op_dir):
    """Run accuracy pytest for op on gpu_id and save log."""
    print(f"[INFO][GPU {gpu_id}] Starting accuracy for '{op}'")
    env = os.environ.copy()
    env["CUDA_VISIBLE_DEVICES"] = str(gpu_id)
    cmd = f'pytest -m "{op}" --ref cpu'
    out, err, code = run_cmd_capture(cmd, cwd=os.path.join(flaggems_path, "tests"), env=env)
    combined = out + "\n" + err
    acc_log = os.path.join(op_dir, "accuracy.log")
    with open(acc_log, "w") as f:
        f.write(combined)
    print(f"[INFO][GPU {gpu_id}] Saved accuracy log: {acc_log}")
    passed, failed, skipped, xfailed, xpassed, errors, total = parse_pytest_summary_from_text(combined)
    status = "PASS" if (code == 0 and failed == 0 and errors == 0) else "FAIL"
    return {
        "passed": passed,
        "failed": failed,
        "skipped": skipped,
        "xfailed": xfailed,
        "xpassed": xpassed,
        "errors": errors,
        "total": total,
        "status": status,
        "log_path": acc_log,
        "exit_code": code
    }

def run_benchmark_and_parse(op, gpu_id, flaggems_path, op_dir):
    """Run benchmark pytest for op, parse multiple performance rows."""
    print(f"[INFO][GPU {gpu_id}] Starting benchmark for '{op}'")
    env = os.environ.copy()
    env["CUDA_VISIBLE_DEVICES"] = str(gpu_id)
    benchmark_dir = os.path.join(flaggems_path, "benchmark")
    ensure_dir(benchmark_dir)
    # remove old benchmark files
    pattern = f"result-m_{op}--level_core--record_log.log"
    for p in Path(benchmark_dir).glob(pattern):
        try: p.unlink()
        except: pass
    # run benchmark
    cmd = f'pytest -m "{op}" --level core --record log'
    out, err, code = run_cmd_capture(cmd, cwd=benchmark_dir, env=env)
    combined = out + "\n" + err
    perf_console_log = os.path.join(op_dir, "perf.log")
    with open(perf_console_log, "w") as f: f.write(combined)
    # locate result file
    perf_result_file = None
    for p in Path(benchmark_dir).glob(pattern):
        perf_result_file = str(p)
        break
    if not perf_result_file:
        print(f"[WARN] Benchmark result file not found for op '{op}'")
        return {"status":"NO_RESULT","perf_console_log":perf_console_log,"perf_result_file":None,"parsed_summary":None,"performance_rows":[]}
    # move to op_dir
    dest = os.path.join(op_dir, os.path.basename(perf_result_file))
    shutil.move(perf_result_file, dest)
    perf_result_file = dest
    # dedupe
    with open(perf_result_file,"r") as f: lines=f.readlines()
    seen=set(); unique=[]
    for l in lines:
        if l not in seen: seen.add(l); unique.append(l)
    with open(perf_result_file,"w") as f: f.writelines(unique)
    # run summary_for_plot.py -> parsed_summary.log
    parsed_summary_path = os.path.join(op_dir, "parsed_summary.log")
    try:
        cmd = f'python3 summary_for_plot.py "{perf_result_file}"'
        out2, err2, code2 = run_cmd_capture(cmd, cwd=benchmark_dir)
        combined2 = out2 + "\n" + err2
        with open(parsed_summary_path,"w") as f: f.write(combined2)
        # dedupe
        plines=[l.rstrip() for l in combined2.splitlines() if l.strip()!=""]
        header_idx=-1
        for i,l in enumerate(plines):
            if l.lower().startswith("op_name"): header_idx=i; break
        header=None; data_lines=[]
        if header_idx>=0: header=plines[header_idx]; data_lines=[l for j,l in enumerate(plines) if j!=header_idx]
        else: data_lines=plines
        # unique
        seen2=set(); data_lines2=[]
        for l in data_lines:
            if l not in seen2: seen2.add(l); data_lines2.append(l)
        with open(parsed_summary_path,"w") as f:
            if header: f.write(header+"\n")
            for l in data_lines2: f.write(l+"\n")
    except Exception as e:
        print(f"[WARN] Failed to run summary_for_plot.py: {e}")
    # parse multiple rows
    performance_rows=[]
    try:
        with open(parsed_summary_path,"r") as f:
            lines=f.readlines()
        start=False
        for line in lines:
            line=line.strip()
            if line=="" or line.startswith("XCCL"): continue
            if line.lower().startswith("op_name"): start=True; continue
            if not start: continue
            cols=re.split(r'\s+', line)
            if len(cols)<8: continue
            def clean(x): return x.strip().strip('"').strip("'")
            row = {
                "func_name": clean(cols[0]),
                "float16": clean(cols[1]),
                "float32": clean(cols[2]),
                "bfloat16": clean(cols[3]),
                "int16": clean(cols[4]),
                "int32": clean(cols[5]),
                "bool": clean(cols[6]),
                "cfloat": clean(cols[7]),
            }
            # compute avg_speedup
            vals=[]
            for v in [row["float16"],row["float32"],row["bfloat16"],row["int16"],row["int32"],row["bool"],row["cfloat"]]:
                try: dv=to_decimal(v); vals.append(dv) if dv>0 else None
                except: pass
            if vals: row["avg_speedup"]=str(round(float(sum(vals)/len(vals)),6))
            else: row["avg_speedup"]="0"
            performance_rows.append(row)
    except Exception as e:
        print(f"[WARN] Parsing parsed_summary.log failed: {e}")

    return {
        "status":"OK",
        "perf_console_log":perf_console_log,
        "perf_result_file":perf_result_file,
        "parsed_summary":parsed_summary_path,
        "performance_rows":performance_rows
    }

# ---------------- Per-GPU worker ----------------

def worker_process_ops(gpu_id, ops_list, flaggems_path, results_dir):
    results={}
    for op in ops_list:
        op_clean=op.strip()
        if op_clean=="": continue
        op_dir=os.path.join(results_dir, op_clean)
        ensure_dir(op_dir)
        print(f"[INFO][GPU {gpu_id}] Processing operator '{op_clean}' in {op_dir}")
        acc=run_accuracy(op_clean,gpu_id,flaggems_path,op_dir)
        perf=run_benchmark_and_parse(op_clean,gpu_id,flaggems_path,op_dir)
        results[op_clean]={"gpu":gpu_id,"accuracy":acc,"performance":perf}
        print(f"[INFO][GPU {gpu_id}] Finished operator '{op_clean}' (acc:{acc['status']} perf:{perf.get('status','N/A')})")
    return results

# ---------------- Top-level orchestration ----------------

def distribute_ops_round_robin(ops,gpus):
    tasks={g:[] for g in gpus}
    for i,op in enumerate(ops): tasks[gpus[i%len(gpus)]].append(op)
    return tasks

def write_summary_json_and_xlsx(summary_map, results_dir):
    """
    Write summary.json and summary.xlsx based on summary_map.

    - Each op in summary_map has:
        - accuracy: dict
        - performance: dict with performance_rows: list of dicts
          Each dict contains: func_name, float16, float32, bfloat16, int16, int32, bool, cfloat, avg_speedup
    - Excel: merge accuracy columns across multiple performance rows
    """
    # JSON
    summary_list = []
    for op, info in summary_map.items():
        entry = {
            "operator": op,
            "accuracy": info.get("accuracy"),
            "performance": info.get("performance").get("performance_rows", [])
        }
        summary_list.append(entry)

    json_path = os.path.join(results_dir, "summary.json")
    with open(json_path, "w") as f:
        json.dump(summary_list, f, indent=2)
    print(f"[INFO] Wrote JSON summary: {json_path}")

    # Excel
    xlsx_path = os.path.join(results_dir, "summary.xlsx")
    wb = Workbook()
    ws = wb.active
    ws.title = "Summary"

    header = [
        "operator","acc_status","passed","failed","skipped","xfailed","xpassed","errors","total","acc_exit_code",
        "func_name","avg_speedup","float16","float32","bfloat16","int16","int32","bool","cfloat",
        "perf_status","perf_console_log","perf_result_file","parsed_summary"
    ]
    ws.append(header)

    for op, info in summary_map.items():
        acc = info.get("accuracy", {})
        perf = info.get("performance", {})
        perf_rows = perf.get("performance_rows", [])
        if not perf_rows:
            perf_rows = [{}]  # fallback if no performance rows

        first = True
        for row in perf_rows:
            ws.append([
                op if first else "",  # merge operator for first row
                acc.get("status") if first else "",
                acc.get("passed") if first else "",
                acc.get("failed") if first else "",
                acc.get("skipped") if first else "",
                acc.get("xfailed") if first else "",
                acc.get("xpassed") if first else "",
                acc.get("errors") if first else "",
                acc.get("total") if first else "",
                acc.get("exit_code") if first else "",
                row.get("func_name",""),
                row.get("avg_speedup",""),
                row.get("float16",""),
                row.get("float32",""),
                row.get("bfloat16",""),
                row.get("int16",""),
                row.get("int32",""),
                row.get("bool",""),
                row.get("cfloat",""),
                perf.get("status",""),
                perf.get("perf_console_log",""),
                perf.get("perf_result_file",""),
                perf.get("parsed_summary","")
            ])
            first = False

    wb.save(xlsx_path)
    print(f"[INFO] Wrote XLSX summary: {xlsx_path}")

    return json_path, xlsx_path

# ---------------- CLI / Main ----------------

def main():
    parser=argparse.ArgumentParser()
    parser.add_argument("--flaggems",required=True,help="Root path of FlagGems")
    parser.add_argument("--op-list",required=True,help="Path to ops list file (one op per line, # comments allowed)")
    parser.add_argument("--gpus",default="0",help="Comma-separated GPU ids, e.g. '0,1,2,3'")
    parser.add_argument("--results-dir",default=None,help="Optional: base results dir; if omitted a timestamped one under current dir will be used")
    args=parser.parse_args()

    FLAGGEMS_PATH=args.flaggems
    OP_LIST=args.op_list
    GPUS=[int(x.strip()) for x in args.gpus.split(",") if x.strip()!=""]
    if len(GPUS)==0: raise SystemExit("No GPUs provided")

    # read operators
    ops=[]
    with open(OP_LIST,"r") as f:
        for line in f:
            line2=line.strip()
            if line2=="" or line2.startswith("#"): continue
            ops.append(line2)
    if len(ops)==0: raise SystemExit("No operators found in op list")

    # results dir
    if args.results_dir: RESULTS_DIR=args.results_dir
    else: RESULTS_DIR=os.path.join(os.path.dirname(os.path.abspath(__file__)),f"results_{now_ts()}")
    ensure_dir(RESULTS_DIR)
    print(f"[INFO] Results directory: {RESULTS_DIR}")

    # distribute
    tasks=distribute_ops_round_robin(ops,GPUS)
    print(f"[INFO] Distributing {len(ops)} ops across GPUs: {GPUS}")
    for g,lst in tasks.items(): print(f"[INFO] GPU {g} <- {len(lst)} ops")

    # run per-GPU workers in parallel
    overall_results={}
    with ThreadPoolExecutor(max_workers=len(GPUS)) as exe:
        future_map={}
        for g in GPUS:
            op_list=tasks.get(g,[])
            if op_list:
                future=exe.submit(worker_process_ops,g,op_list,FLAGGEMS_PATH,RESULTS_DIR)
                future_map[future]=g
        for fut in as_completed(future_map):
            g=future_map[fut]
            try:
                res=fut.result()
                overall_results.update(res)
                print(f"[INFO] GPU {g} finished its assigned ops")
            except Exception as e:
                print(f"[ERROR] Worker for GPU {g} raised exception: {e}")

    # write final summary
    json_path,xlsx_path=write_summary_json_and_xlsx(overall_results,RESULTS_DIR)
    print(f"[INFO] Final summary written to: {json_path}, {xlsx_path}")
    print("[INFO] All done.")

if __name__=="__main__":
    main()
